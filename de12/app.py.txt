def normalized(value):
    return "".join(
        char for char in unicodedata.normalize("NFKC", value).casefold()
        if char.isalnum()
    )


@lru_cache(maxsize=64)
def search_music(title, artist):
    query = urlencode({
        "term": f"{title} {artist}",
        "country": "JP",
        "media": "music",
        "entity": "song",
        "limit": 20,
    })
    with urlopen("https://itunes.apple.com/search?" + query, timeout=5) as response:
        tracks = json.load(response).get("results", [])

    for track in tracks:
        same_title = normalized(track.get("trackName", "")) == normalized(title)
        same_artist = normalized(track.get("artistName", "")) == normalized(artist)
        if same_title and same_artist:
            return track
    return {}

def show(view, **values):
    return render_template_string(
        PAGE, view=view, kept=session.get("kept", []), member_icons=MEMBER_ICONS, **values
    )


@app.get("/")
def home():
    return redirect(url_for("step", number=0))


@app.route("/step/<int:number>", methods=["GET", "POST"])
def step(number):
    if number >= len(STEPS):
        abort(404)
    for previous in range(number):
        if STEPS[previous][0] not in session:
            return redirect(url_for("step", number=previous))

    key, _, question, options = STEPS[number]
    if request.method == "POST":
        choice = request.form.get("choice")
        if choice not in options:
            abort(400)
        session[key] = choice
        if number == len(STEPS) - 1:
            return redirect(url_for("results"))
        return redirect(url_for("step", number=number + 1))

    return show("step", number=number, question=question,
                options=options, selected=session.get(key))


@app.get("/results")
def results():
    for number, (key, _, _, _) in enumerate(STEPS):
        if key not in session:
            return redirect(url_for("step", number=number))

    member, mood, pitch = (session[key] for key in ("member", "mood", "pitch"))
    matches = [
        song for song in SONGS.values()
        if song["mood"] == mood and song["pitch"] == pitch
    ]
    matches.sort(key=lambda song: song["member"] != member)
    return show("results", songs=matches, member=member, mood=mood, pitch=pitch)


@app.get("/song/<song_id>")
def detail(song_id):
    song = SONGS.get(song_id)
    if song is None:
        abort(404)
    try:
        track = search_music(song["title"], song["artist"])
    except (OSError, ValueError):
        track = {}
    return show("detail", song=song, track=track)


@app.post("/keep/<song_id>")
def keep(song_id):
    if song_id not in SONGS:
        abort(404)
    kept = session.get("kept", [])
    if song_id not in kept:
        session["kept"] = kept + [song_id]
    return redirect(url_for("detail", song_id=song_id))


@app.get("/queue")
def queue():
    songs = [SONGS[song_id] for song_id in session.get("kept", []) if song_id in SONGS]
    return show("queue", songs=songs, favorites=session.get("favorites", []))


@app.post("/favorite/<song_id>")
def toggle_favorite(song_id):
    if song_id not in SONGS or song_id not in session.get("kept", []):
        abort(404)
    favorites = session.get("favorites", [])
    if song_id in favorites:
        session["favorites"] = [item for item in favorites if item != song_id]
    else:
        session["favorites"] = favorites + [song_id]
    return redirect(url_for("queue"))
